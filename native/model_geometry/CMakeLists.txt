cmake_minimum_required(VERSION 3.18)
project(optimizer_geometry LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(MANIFOLD_TEST OFF CACHE BOOL "" FORCE)
set(MANIFOLD_PYBIND OFF CACHE BOOL "" FORCE)
set(MANIFOLD_CBIND OFF CACHE BOOL "" FORCE)
set(MANIFOLD_CROSS_SECTION OFF CACHE BOOL "" FORCE)
set(MANIFOLD_PAR OFF CACHE BOOL "" FORCE)
set(MANIFOLD_NO_IOSTREAM ON CACHE BOOL "" FORCE)
include(FetchContent)
FetchContent_Declare(manifold URL https://github.com/elalish/manifold/archive/ce50d78021d64507f89e8c9fc2c2e51018117857.tar.gz)
FetchContent_MakeAvailable(manifold)
if(ANDROID)
  add_compile_options(-fexceptions -frtti)
  add_library(optimizer_geometry SHARED geometry_core.cpp geometry_jni.cpp)
  target_link_libraries(optimizer_geometry PRIVATE manifold)
  target_link_options(optimizer_geometry PRIVATE -Wl,-z,max-page-size=16384 -Wl,-z,common-page-size=16384)
else()
  add_executable(geometry_test geometry_test.cpp geometry_core.cpp)
  target_link_libraries(geometry_test PRIVATE manifold)
  enable_testing()
  add_test(NAME geometry_cut_pins COMMAND geometry_test)
endif()
